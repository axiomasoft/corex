#!/usr/bin/env python3
"""Match a finite, decision-bound unpublished API delta; unknown breaks fail closed."""
import argparse
import hashlib
import json
from pathlib import Path
import sys


def admitted(root, package, base, errors):
    carrier = json.loads((root / 'scripts/bc/approved-deltas.json').read_text())
    if carrier.get('schema') != 'corex-bc-approved-deltas/1':
        return False
    rule = carrier.get('packages', {}).get(package)
    if not isinstance(rule, dict) or rule.get('base_revision') != base:
        return False
    decisions = rule.get('decisions')
    if not isinstance(decisions, list) or not decisions:
        return False
    for decision in decisions:
        path = (root / decision['path']).resolve()
        if not path.is_relative_to(root.resolve()) or not path.is_file():
            return False
        if hashlib.sha256(path.read_bytes()).hexdigest() != decision['sha256']:
            return False
        source = path.read_text()
        if '\nstatus: accepted\n' not in source:
            return False
    def signature(row):
        return (hashlib.sha256(row['description'].encode()).hexdigest(), row['path'], row['modificationType'])
    actual = {signature(row) for row in errors}
    approved = {(row['description_sha256'], row['path'], row['modificationType']) for row in rule.get('changes', [])}
    return bool(actual) and actual == approved


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, required=True)
    parser.add_argument('--package', required=True)
    parser.add_argument('--base', required=True)
    args = parser.parse_args()
    try:
        reports = [json.loads(line)['errors'] for line in sys.stdin if line.startswith('{"errors":')]
        return 0 if len(reports) == 1 and admitted(args.root, args.package, args.base, reports[0]) else 1
    except (OSError, ValueError, KeyError, TypeError):
        return 1


if __name__ == '__main__':
    raise SystemExit(main())

```php
            $context = new TenantContext($current);
            foreach ($this->bootstrappers as $bootstrapper) {
                $this->started[] = $bootstrapper;
                $bootstrapper->bootstrap($context, $database);
            }

            $this->account = $current;
            $this->workspace = null;
            $this->resetScope();
            $this->events->dispatch(new TenantContextInitialized($current->id));
        } catch (Throwable $failure) {
            $this->account = null;
            $this->workspace = null;
            try {
                $this->revertAll(array_reverse($this->started));
            } catch (Throwable $cleanup) {
                $this->reportCleanupFailure($failure, $cleanup);
            } finally {
                $this->started = [];
                $this->resetScope();
            }

            throw $failure;
        }
    }

    #[Override]
    public function setWorkspace(?WorkspaceRef $workspace): void
    {
        $this->assertHealthy();
        if ($workspace !== null && $this->account === null) {
            throw TenantContextUnavailable::notInitialized();
        }
        $this->workspace = $workspace;
    }

    #[Override]
    public function end(): void
    {
        $this->assertHealthy();
        $accountId = $this->account?->id;
        $this->account = null;
        $this->workspace = null;

        try {
            $this->revertAll(array_reverse($this->started));
        } finally {
            $this->started = [];
            $this->resetScope();
        }
        if ($accountId !== null) {
            $this->events->dispatch(new TenantContextEnded($accountId));
        }
    }

    #[Override]
    public function runFor(AccountRef $account, ?WorkspaceRef $workspace, Closure $callback): mixed
    {
        $previousAccount = $this->account;
        $previousWorkspace = $this->workspace;
        $original = null;

        try {
            $this->initialize($account);
            $this->setWorkspace($workspace);
            return $callback();
        } catch (Throwable $failure) {
            $original = $failure;
            throw $failure;
        } finally {
            // A failed revert poisons the runtime; restoration must not mask its causes.
            if (! $this->poisoned) {
                try {
                    if ($previousAccount === null) {
                        $this->end();
                    } else {
                        $this->initialize($previousAccount);
                        $this->setWorkspace($previousWorkspace);
                    }
                } catch (Throwable $cleanup) {
                    if ($original !== null) {
                        $this->reportCleanupFailure($original, $cleanup);
                    }
                    throw $cleanup;
                }
            }
        }
    }

    private function reportCleanupFailure(Throwable $original, Throwable $cleanup): never
    {
        try {
            $this->logger->error('Tenant runtime cleanup failed', [
                'exception' => $cleanup,
                'original_exception' => $original,
            ]);
        } catch (Throwable) {
            // A failing log transport must not replace either runtime cause.
        }
        throw new TenantCleanupFailure($original, $cleanup);
    }

    /** @param list<TenantBootstrapper> $bootstrappers already in revert order */
    private function revertAll(array $bootstrappers): void
    {
        $first = null;

        foreach ($bootstrappers as $bootstrapper) {
            try {
                $bootstrapper->revert();
            } catch (Throwable $failure) {
                $first ??= $failure;
            }
        }

        if ($first !== null) {
            $this->poisoned = true;
            throw $first;
        }
    }

    private function resetScope(): void
    {
        $this->container->forgetScopedInstances();
    }

    private function assertHealthy(): void
    {
        if ($this->poisoned) {
            throw TenantContextUnavailable::runtimePoisoned();
        }
    }
}
```

Worked example (в `TenancyEngineTest`): `initialize(A)` → `context()->account->id === 'A'`, default connection `tenant`, database = `db_name` A;
`runFor(B, ws, fn)` внутри → в fn `context()->account->id === 'B'`, workspace `ws`; после → снова A, workspace прежний;
исключение в fn → то же восстановление; `end()` → default connection = central, `context() === null`;
bootstrapper №2 мутирует facility и бросает при `initialize` → №2 и №1 откатаны в обратном порядке, scoped-кэш очищен, `initialized() === false`.
Отдельно: `runFor(missing)` из A/workspace W, resolver failure и `TenantContextInitialized` listener failure восстанавливают A/W, callback не запускается.
`revert()` failure: остальные revert выполняются, scoped-кэш очищается; manager помечен poisoned и все следующие context/initialized/initialize/end/setWorkspace бросают `runtimePoisoned()`.
Этот отказ не считается восстановленным central: resident host обязан завершить текущий процесс/пересоздать приложение перед следующим запросом/job; повторный `runFor` не разрешён. При двойном отказе TenantCleanupFailure сохраняет исходную ошибку в getPrevious(), cleanup — в readonly property; PSR logger получает обе причины. Отказ log transport не маскирует их. runFor не пытается восстановить poisoned runtime и сохраняет это исключение. RuntimeRegistrar передаёт существующий PSR LoggerInterface (psr/log уже declared); full provider-boot test проверяет binding.
Bootstrapper не зависит от `TenancyManager` или `TenantContextResolver`, прямо либо транзитивно; событие не инициирует reentrant switch. Options и DI-тест проверяют штатный граф; SPI документирует эти ограничения для host-реализаций.

Единственная точка cleanup — `JobAttempted`, отправляемый в `finally` как Worker::process, так и SyncQueue::executeJob (установленный Laravel). `JobFailed`, `JobProcessed`, `JobExceptionOccurred` не завершают контекст: failed callbacks ещё могут работать в контексте job. `QueueTenantContext` singleton экземпляр без static-состояния, stack рамок с job object identity.

```php
final class QueueTenantContext
{
    private array $frames = [];

    public function __construct(
        private readonly TenancyManager $manager,
        private readonly AccountRead $accounts,
        private readonly \Closure $workspaceResolver, // (?id) -> ?WorkspaceRef, tenant query after initialize
        private readonly string $centralConnection,
    ) {}

    public function processing(\Illuminate\Queue\Events\JobProcessing $event): void
    {
        $restore = $event->job instanceof \Illuminate\Queue\Jobs\SyncJob || $this->frames !== [];
        $previous = $restore ? $this->manager->context() : null;
        $this->frames[] = ['job' => $event->job, 'previous' => $previous]; // before end/find/initialize
        $this->manager->end();
        $payload = $event->job->payload()['corex_tenant'] ?? null;
        if ($payload === null) {
            return; // central job; sync caller is restored only after this job finishes
        }
        if (! is_array($payload) || ! is_string($payload['account'] ?? null)
            || ! array_key_exists('workspace', $payload)
            || ($payload['workspace'] !== null && ! is_string($payload['workspace']))) {
            throw TenantContextUnavailable::invalidPayload();
        }
        $ref = $this->accounts->find($payload['account'], $this->centralConnection)
            ?? throw TenantContextUnavailable::missingAccount($payload['account']);
        $this->manager->initialize($ref);
        if ($payload['workspace'] !== null) {
            $workspace = ($this->workspaceResolver)($payload['workspace'])
                ?? throw TenantContextUnavailable::missingWorkspace($payload['workspace']);
            $this->manager->setWorkspace($workspace);
        }
    }

    public function finished(\Illuminate\Queue\Events\JobAttempted $event): void
    {
        $frame = end($this->frames);
        if ($frame === false || $frame['job'] !== $event->job) {
            return; // unrelated/repeated completion cannot pop another job's frame
        }
        array_pop($this->frames);
        $this->manager->end();
        if ($frame['previous'] !== null) {
            $this->manager->initialize($frame['previous']->account);
            $this->manager->setWorkspace($frame['previous']->workspace);
        }
    }
}
```

Factory workspaceResolver использует `Workspace::query()->find($id)` и создаёт `WorkspaceRef(id, slug, type, path, parentId)`. Она вызывается только после initialize. Async top-level игнорирует оставшийся контекст до job и после завершения всегда central; отсутствующий account не исполняет job в старой базе. Sync, включая вложенный sync, сохраняет account/workspace вызывающего кода и восстанавливает его после успешного или неуспешного завершения. Ошибка cleanup/restoration пробрасывается, worker/request прекращается; poisoned manager требует перезапуска приложения. Queue retry сохраняет исходный `corex_tenant`; отдельный `JobRetryRequested` listener отсутствует.

### 6c. `AccountStatusGate` без hydrated Stancl Account (D102)

AccountRef не расширяется owner_identity_id. Owner читается через существующий AccountRead::ownerIdentityId(id, centralConnection), явно central. Не-null authenticated user и строгое совпадение id обязательны; null owner никогда не разрешает. Статусы и payment/export/read-only routes сохраняют прежнюю политику.

```php
<?php

declare(strict_types=1);

namespace CoreX\Tenancy\Laravel\Http\Middleware;

use Closure;
use CoreX\Attributes\Spi;
use CoreX\Attributes\SpiKind;
use CoreX\Context\Data\AccountRef;
use CoreX\Context\Contracts\TenantContextResolver;
use CoreX\Tenancy\Account\Contracts\AccountRead;
use Illuminate\Contracts\Auth\Factory as AuthFactory;
use CoreX\Tenancy\Laravel\Support\StatusGatePolicy;
use Illuminate\Http\Request;
use Illuminate\Http\Response;

/**
 * B-11 §5.1 access rules, status read from AccountRef; owner identity read through the central AccountRead port. Fail-closed: `match` carries no default-pass
 * branch, an unknown status (including `pending`/`provisioning`, which
 * should never reach this middleware — cache/domain already reject them)
 * falls through to 403.
 */
#[Spi(kind: SpiKind::Call, since: '1.0')]
final class AccountStatusGate
{
    public function __construct(
        private readonly StatusGatePolicy $statusGate,
        private readonly TenantContextResolver $contexts,
        private readonly AccountRead $accounts,
        private readonly AuthFactory $auth,
        private readonly string $centralConnection,
    ) {}

    /** @return Response|mixed */
    public function handle(Request $request, Closure $next): mixed
    {
        $account = $this->contexts->current()->account;

        return match ($account->status) {
            'trial', 'active' => $next($request),
            'suspended' => $this->ownerOnRoute(
                $request,
                $next,
                $account,
                $this->statusGate->paymentRoute,
            ),
            'grace' => $this->graceGate($request, $next, $account),
            'exporting', 'deleted' => abort(410),
            default => abort(403),
        };
    }

    /** @return Response|mixed */
    private function ownerOnRoute(
        Request $request,
        Closure $next,
        AccountRef $account,
        ?string $route,
    ): mixed {
        if (! $this->isOwner($account) || ! $this->matchesConfiguredRoute($request, $route)) {
            abort(403);
        }

        return $next($request);
    }

    /** @return Response|mixed */
    private function graceGate(Request $request, Closure $next, AccountRef $account): mixed
    {
        if (! $this->isOwner($account)) {
            abort(403);
        }

        $readOnly = $this->statusGate->isReadOnlyMethod($request->method());
        $onExportRoute = $this->matchesConfiguredRoute($request, $this->statusGate->exportRoute);

        if (! $readOnly && ! $onExportRoute) {
            abort(403);
        }

        return $next($request);
    }

    private function isOwner(AccountRef $account): bool
    {
        $userId = $this->auth->guard()->id();

        return $userId !== null && $this->accounts->ownerIdentityId($account->id, $this->centralConnection) === $userId;
    }

    private function matchesConfiguredRoute(Request $request, mixed $routeConfig): bool
    {
        if (! is_string($routeConfig) || $routeConfig === '') {
            return false;
        }

        if ($request->route()?->getName() === $routeConfig) {
            return true;
        }

        return $request->is(ltrim($routeConfig, '/'));
    }
}
```
